<script>
const CFG = <?!= cfg ?>;

const NOMES = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday'];
const CURTOS = ['MON', 'TUE', 'WED', 'THU', 'FRI'];
const MESES = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

let ses = null;                          // { token, email, nome, papel, taxa, ate, equipe }
let alvo = null;                         // timesheet sendo visto
let segunda = segundaDe(new Date());
let dados = {};
let pendente = null;

function meuProprio() { return ses && alvo === ses.email; }
function taxaDoAlvo() {
  if (!ses) return 0;
  if (meuProprio()) return ses.taxa;
  const p = (ses.equipe || []).filter(function (x) { return x.email === alvo; })[0];
  return p ? p.taxa : 0;
}

// ---------- Datas ----------
function segundaDe(d) {
  const x = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  x.setDate(x.getDate() - ((x.getDay() + 6) % 7));
  return x;
}
function somaDias(d, n) {
  const x = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  x.setDate(x.getDate() + n);
  return x;
}
function iso(d) {
  return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
}
function deIso(s) { const p = String(s).split('-'); return new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2])); }
function curto(d) { return d.getDate() + ' ' + MESES[d.getMonth()]; }
function periodo(i) { return curto(i) + ' – ' + curto(somaDias(i, 4)) + ' ' + i.getFullYear(); }

// ---------- Cálculo ----------
function paraMin(hhmm) {
  const p = String(hhmm || '').split(':');
  if (p.length !== 2) return null;
  const h = parseInt(p[0], 10), m = parseInt(p[1], 10);
  if (isNaN(h) || isNaN(m)) return null;
  return h * 60 + m;
}
function minutosDoDia(d) {
  d = d || {};
  const ini = paraMin(d.entrada), fim = paraMin(d.saida);
  if (ini === null || fim === null) return 0;
  let total = (fim <= ini ? fim + 1440 : fim) - ini;
  if (d.lunch) total -= CFG.lunch;
  if (d.smoko) total -= CFG.smoko;
  return Math.max(0, total);
}
function totaisDaSemana(inicio, registro) {
  return [0,1,2,3,4].map(function (i) { return minutosDoDia((registro || {})[iso(somaDias(inicio, i))]); });
}
function hm(min) { return Math.floor(min / 60) + ':' + String(min % 60).padStart(2, '0'); }
function dec(min) { return (min / 60).toFixed(2) + ' h'; }
function dinheiro(min, taxa) { return CFG.moeda + (min / 60 * (taxa || 0)).toFixed(2); }
function soma(l) { return l.reduce(function (a, b) { return a + b; }, 0); }
function faixaMini(t) {
  return t.map(function (min, i) {
    return '<div><span>' + CURTOS[i] + '</span><b>' + (min ? hm(min) : '—') + '</b></div>';
  }).join('');
}
function vazioDia(d) { return !d || (!d.entrada && !d.saida && !d.lunch && !d.smoko); }
function resumoDia(d) {
  if (vazioDia(d)) return '—';
  const e = [];
  if (d.smoko) e.push('smoko');
  if (d.lunch) e.push('lunch');
  return (d.entrada || '--:--') + '–' + (d.saida || '--:--') + (e.length ? ' (' + e.join(', ') + ')' : '');
}

// ---------- Semana ----------
function desenhar() {
  document.getElementById('titulo').firstChild.nodeValue = periodo(segunda).slice(0, -5) + ' ';
  document.getElementById('subtitulo').textContent = segunda.getFullYear();

  const hojeIso = iso(new Date());
  const caixa = document.getElementById('dias');
  const podeEditar = meuProprio();
  caixa.innerHTML = '';

  NOMES.forEach(function (nome, i) {
    const data = somaDias(segunda, i);
    const chave = iso(data);
    const d = dados[chave] || { entrada: '', saida: '', lunch: false, smoko: false };
    const min = minutosDoDia(d);
    const viraDia = paraMin(d.entrada) !== null && paraMin(d.saida) !== null && paraMin(d.saida) <= paraMin(d.entrada);
    const trava = podeEditar ? '' : ' disabled';

    const el = document.createElement('div');
    el.className = 'dia' + (chave === hojeIso ? ' hoje' : '');
    el.innerHTML =
      '<div class="dia-topo">' +
        '<div class="dia-nome">' + nome + '<small>' + curto(data) + '</small></div>' +
        '<div class="dia-total' + (min ? '' : ' zero') + '">' + (min ? hm(min) : '—') + '</div>' +
      '</div>' +
      '<div class="horas">' +
        '<input type="time" value="' + d.entrada + '" data-campo="entrada" aria-label="Start ' + nome + '"' + trava + '>' +
        '<span class="ate">to</span>' +
        '<input type="time" value="' + d.saida + '" data-campo="saida" aria-label="Finish ' + nome + '"' + trava + '>' +
      '</div>' +
      '<div class="chips">' +
        '<label class="chip' + (d.smoko ? ' on' : '') + (podeEditar ? '' : ' travado') + '">' +
          '<input type="checkbox" data-campo="smoko"' + (d.smoko ? ' checked' : '') + trava + '>' +
          '<span class="marcador"></span>Smoko' + (CFG.smoko ? ' −' + CFG.smoko + ' min' : '') +
        '</label>' +
        '<label class="chip' + (d.lunch ? ' on' : '') + (podeEditar ? '' : ' travado') + '">' +
          '<input type="checkbox" data-campo="lunch"' + (d.lunch ? ' checked' : '') + trava + '>' +
          '<span class="marcador"></span>Lunch −' + CFG.lunch + ' min' +
        '</label>' +
      '</div>' +
      (viraDia ? '<div class="aviso">Shift crosses midnight — counted into the next day.</div>' : '');

    el.querySelectorAll('input').forEach(function (inp) {
      inp.addEventListener('change', function () {
        const anterior = dados[chave] ? JSON.parse(JSON.stringify(dados[chave])) : null;
        const novo = anterior ? JSON.parse(JSON.stringify(anterior)) : { entrada: '', saida: '', lunch: false, smoko: false };
        novo[inp.dataset.campo] = (inp.type === 'checkbox') ? inp.checked : inp.value;

        if (vazioDia(anterior)) {
          enviarDia(chave, novo, '');
        } else {
          pendente = { dia: chave, valores: novo };
          document.getElementById('textoMotivo').textContent =
            'Changing ' + nome + ' ' + curto(data) + ': ' + resumoDia(anterior) + ' → ' + resumoDia(novo) + '.';
          document.getElementById('campoMotivo').value = '';
          document.getElementById('erroMotivo').textContent = '';
          abrirJanela('janelaMotivo');
          document.getElementById('campoMotivo').focus();
        }
      });
    });

    caixa.appendChild(el);
  });

  const totais = totaisDaSemana(segunda, dados);
  document.getElementById('diasMini').innerHTML = faixaMini(totais);
  document.getElementById('semanaHM').textContent = hm(soma(totais));
  document.getElementById('semanaDec').textContent = dec(soma(totais));
  document.getElementById('semanaPag').textContent = dinheiro(soma(totais), taxaDoAlvo());
  document.getElementById('btnLimpar').classList.toggle('oculto', !podeEditar);
  document.getElementById('lendoAlguem').classList.toggle('oculto', podeEditar);
}

function enviarDia(dia, valores, motivo) {
  aviso('Saving…');
  google.script.run
    .withSuccessHandler(function (r) {
      dados[dia] = valores;
      if (vazioDia(valores)) delete dados[dia];
      desenhar();
      aviso('Saved at ' + r.hora);
    })
    .withFailureHandler(function (e) {
      aviso(e.message, true);
      if (/session expired/i.test(e.message)) return sairDaSessao();
      carregar();
    })
    .salvarDia(ses.token, iso(segunda), dia, valores, motivo);
}

// ---------- Histórico ----------
function abrirHistorico() {
  const painel = document.getElementById('painelHistorico');
  painel.innerHTML = '<div class="vazio">Loading…</div>';

  google.script.run
    .withSuccessHandler(function (lista) {
      if (!lista || !lista.length) { painel.innerHTML = '<div class="vazio">No weeks saved yet.</div>'; return; }
      const atualIso = iso(segundaDe(new Date()));
      const taxa = taxaDoAlvo();

      painel.innerHTML = lista.map(function (item) {
        const inicio = deIso(item.segunda);
        const t = totaisDaSemana(inicio, item.dados);
        return '<div class="cartao">' +
            '<div class="topo">' +
              '<div class="periodo">' + periodo(inicio) + (item.segunda === atualIso ? '<span class="atual">CURRENT</span>' : '') + '</div>' +
              '<div class="soma">' + hm(soma(t)) + '<em>' + dinheiro(soma(t), taxa) + ' gross</em></div>' +
            '</div>' +
            '<div class="mini">' + faixaMini(t) + '</div>' +
            '<button class="btn claro" style="margin-top:10px" data-semana="' + item.segunda + '">Open this week</button>' +
          '</div>';
      }).join('');

      painel.querySelectorAll('[data-semana]').forEach(function (b) {
        b.addEventListener('click', function () {
          segunda = deIso(b.dataset.semana);
          mostrarAba('semana');
          carregar();
        });
      });
    })
    .withFailureHandler(function (e) { painel.innerHTML = '<div class="vazio">' + e.message + '</div>'; })
    .carregarHistorico(ses.token, alvo);
}

function carregar() {
  aviso('Loading…');
  google.script.run
    .withSuccessHandler(function (res) { dados = res || {}; desenhar(); aviso(''); })
    .withFailureHandler(function (e) {
      aviso(e.message, true);
      if (/session expired/i.test(e.message)) sairDaSessao();
    })
    .carregarSemana(ses.token, alvo, iso(segunda));
}

function abrirJanela(id) { document.getElementById(id).classList.add('aberta'); }

// ---------- Sessão ----------
function guardarToken(t) {
  try { t ? sessionStorage.setItem('tk', t) : sessionStorage.removeItem('tk'); } catch (e) {}
}

function entrar(s) {
  ses = s;
  alvo = s.email;
  guardarToken(s.token);

  document.getElementById('login').classList.add('oculto');
  document.getElementById('app').classList.remove('oculto');
  document.getElementById('meuNome').textContent = s.nome;

  const selo = document.getElementById('meuPapel');
  selo.textContent = s.papel.toUpperCase();
  selo.className = 'selo' + (s.papel === 'admin' ? ' admin' : '');

  const ehAdmin = s.papel === 'admin';
  document.getElementById('abaEquipe').classList.toggle('oculto', !ehAdmin);
  document.getElementById('escolha').classList.toggle('oculto', !ehAdmin);

  if (ehAdmin) {
    const sel = document.getElementById('selPessoa');
    sel.innerHTML = (s.equipe || []).map(function (p) {
      return '<option value="' + p.email + '">' + p.nome + (p.email === s.email ? ' (you)' : '') + '</option>';
    }).join('');
    sel.value = s.email;
  }

  mostrarAba('semana');
  carregar();
}

function sairDaSessao() {
  if (ses) google.script.run.sair(ses.token);
  ses = null; alvo = null; dados = {};
  guardarToken(null);
  document.getElementById('app').classList.add('oculto');
  document.getElementById('rodape').classList.add('oculto');
  document.getElementById('login').classList.remove('oculto');
  document.getElementById('campoCodigo').classList.add('oculto');
  document.getElementById('btnEntrar').classList.add('oculto');
  document.getElementById('btnOutroEmail').classList.add('oculto');
  document.getElementById('btnEnviarCodigo').classList.remove('oculto');
  document.getElementById('erroLogin').textContent = '';
}

document.getElementById('btnEnviarCodigo').addEventListener('click', function () {
  const b = this;
  const email = document.getElementById('campoEmail').value.trim();
  if (!email) { document.getElementById('erroLogin').textContent = 'Type your e-mail.'; return; }

  b.disabled = true;
  document.getElementById('erroLogin').textContent = '';
  document.getElementById('textoLogin').textContent = 'Sending…';

  google.script.run
    .withSuccessHandler(function () {
      b.disabled = false;
      b.classList.add('oculto');
      document.getElementById('textoLogin').textContent = 'Code sent to ' + email + '. It expires in 10 minutes.';
      document.getElementById('campoEmail').classList.add('oculto');

      campo.classList.remove('oculto');
      campo.value = '';
      campo.focus();
      document.getElementById('btnEntrar').classList.remove('oculto');
      document.getElementById('btnOutroEmail').classList.remove('oculto');
    })
    .withFailureHandler(function (e) {
      b.disabled = false;
      document.getElementById('textoLogin').textContent = 'Sign in with your work e-mail.';
      document.getElementById('erroLogin').textContent = e.message;
    })

});

  if (!meuProprio()) return;
  const motivo = prompt('Clearing the whole week. Reason:');
  if (motivo === null) return;
  if (motivo.trim().length < 3) { aviso('A reason is required.', true); return; }

  google.script.run
    .withSuccessHandler(function () { dados = {}; desenhar(); aviso('Week cleared'); })
    .withFailureHandler(function (e) { aviso(e.message, true); })
    .limparSemana(ses.token, iso(segunda), motivo.trim());

document.getElementById('btnResumo').addEventListener('click', function () {
  const taxa = taxaDoAlvo();
  const nome = meuProprio() ? ses.nome : (document.getElementById('selPessoa').selectedOptions[0] || {}).text || '';
  const linhas = ['Timesheet — ' + nome, periodo(segunda), ''];
  let total = 0;

  NOMES.forEach(function (n, i) {
    const data = somaDias(segunda, i);
    const d = dados[iso(data)] || {};
    const min = minutosDoDia(d);
    total += min;
    linhas.push(n.substring(0, 3) + ' ' + curto(data) + '  ' + (min ? resumoDia(d) + '  ' + hm(min) : '—'));
  });

  linhas.push('', 'Total: ' + hm(total) + ' (' + dec(total) + ')');
  if (taxa) linhas.push('Gross at ' + CFG.moeda + taxa + '/h: ' + dinheiro(total, taxa));

  const campo = document.getElementById('resumoTexto');
  campo.value = linhas.join('\n');
  abrirJanela('janelaResumo');
  campo.focus(); campo.select();
  try { document.execCommand('copy'); aviso('Summary copied'); } catch (e) { aviso('Select the text and copy it'); }
});
